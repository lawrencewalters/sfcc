# get online categories: 
	input: storefront catalog for sandbox or from staging
	output: file with categoryIds pipe separated for use in next XSLT

java -jar /home/lwalters/SaxonHE12-5J/saxon-he-12.5.jar -s:/home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/bm_data/ods-au/catalogs/titleist-storefront-au-nz/catalog.xml -xsl:get-online-categories.xslt > data/categoryIds.txt

# generate trimmed slots
	input: staging site slots.xml, file with online category ids (optional)
	output: sandbox appropriate slots.xml

cat /home/lwalters/github.com/lawrencewalters/sfcc/sandbox-dataset-builder/data/categoryIds.txt | xargs -t java -jar /home/lwalters/SaxonHE12-5J/saxon-he-12.5.jar -s:/home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/tmp/staging/sites/titleist-au-nz/slots.xml -xsl:generate-trimmed-slots.xslt > /home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/bm_data/ods-au/sites/titleist-au-nz/slots.xml

# get the content asset IDs referenced in the new slots
    input: new slots.xml, file with online category ids (optional)
    output: pipe separated list of content asset ids

cat /home/lwalters/github.com/lawrencewalters/sfcc/sandbox-dataset-builder/data/categoryIds.txt | xargs -t java -jar /home/lwalters/SaxonHE12-5J/saxon-he-12.5.jar -s:/home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/bm_data/ods-au/sites/titleist-au-nz/slots.xml -xsl:get-content-ids-from-slots.xslt > data/contentIds.txt


# generate trimmed library

cat data/contentIds.txt | xargs -t java -jar /home/lwalters/SaxonHE12-5J/saxon-he-12.5.jar -s:/home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/tmp/staging/libraries/TitleistSharedLibrary/library.xml -xsl:generate-trimmed-content-library.xslt > /home/lwalters/bitbucket.org/lyonsconsultinggroup/acushnet/bm_data/ods-au/libraries/TitleistSharedLibrary/library.xml
